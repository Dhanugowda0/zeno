pipeline {
    agent any
    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/Dhanugowda0/zeno.git'
            }
        }
        stage('Build Docker Image') {
            steps {
                sh 'docker build -t zeno-assignment .'
            }
        }
        stage('Deploy') {
            steps {
                sh 'docker stop zeno-app || true'
                sh 'docker rm zeno-app || true'
                sh 'docker run -d --name zeno-app -p 3000:3000 zeno-assignment'
            }
        }
    }
}
